   <uses-permission android:name="android.permission.INTERNET" />

   <uses-permission android:name="android.permission.ACCESS_COARSE_LOCATION" />

Si ya existen algunos, no duplicar, solo agregar los que falten.

▶ PASO 8: GUARDAR USER ID EN LOGIN
▶━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Encuentra donde está el código de login (probablemente en un Activity o Fragment)
y agrega esto DESPUÉS de que el usuario se autentica exitosamente:

   val sharedPreferences = context.getSharedPreferences("UserPrefs", Context.MODE_PRIVATE)
   sharedPreferences.edit().putInt("user_id", userId).apply()

IMPORTANTE: Reemplaza "userId" con la variable que contiene el ID del usuario

Ejemplo:
   // Después de autenticar al usuario
   val userId = response.user.id  // O donde sea que obtengas el ID
   val prefs = context.getSharedPreferences("UserPrefs", Context.MODE_PRIVATE)
   prefs.edit().putInt("user_id", userId).apply()
